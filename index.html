Tip: Paste your deployed Google Apps Script Web App URL here once—it saves automatically in this browser.

1. Create a Google Sheet with tab named Ballots • 2. Paste into Extensions → Apps Script & deploy as Web App (Anyone). 
function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('Ballots') || ss.getSheets()[0];
    var data = JSON.parse(e.postData.contents);

    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        'Server Timestamp',
        'Event Type',
        'Confirmation Code',
        'Control Number',
        'Province / Delegation',
        'Provincial Vote',
        'National Vote',
        'Client Timestamp'
      ]);
    }

    sheet.appendRow([
      new Date(),
      data.event || 'MELTA_2026_BALLOT',
      data.confirmationCode || '',
      data.controlNumber || '',
      data.province || '',
      data.provincialVote || '',
      data.nationalVote || '',
      data.timestamp || ''
    ]);

    return ContentService
      .createTextOutput(JSON.stringify({ status: 'ok' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: 'error', message: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
